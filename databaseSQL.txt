-- 1. Tablas sin dependencias
CREATE TABLE estado (
  id_estado SERIAL PRIMARY KEY,
  nom_estado VARCHAR(100) NOT NULL
);

CREATE TABLE pais (
  id_pais SERIAL PRIMARY KEY,
  nom_pais VARCHAR(100) NOT NULL
);

CREATE TABLE roles (
  id_rol SERIAL PRIMARY KEY,
  nom_rol VARCHAR(20) NOT NULL
);

CREATE TABLE persona_fac (
  id_pers_fac SERIAL PRIMARY KEY,
  cedr_pers_fac VARCHAR(15) NOT NULL,
  razon_pers_fac VARCHAR(15) NOT NULL,
  tipo_pers_fac VARCHAR(20) NOT NULL,
  nacion_pers_fac VARCHAR(30) NOT NULL,
  nom_pers_fac VARCHAR(100) NOT NULL,
  ape_pers_fac VARCHAR(100) NOT NULL,
  tel_pers_fac VARCHAR(10),
  cel_pers_fac VARCHAR(10),
  email_pers_fac VARCHAR(100) NOT NULL,
  direc_pers_fac VARCHAR(20) NOT NULL,
  parent_pers_fac VARCHAR(20) NOT NULL
);

CREATE TABLE cuenta_banco (
  id_cuent_Ban SERIAL PRIMARY KEY,
  tipo_cuent_Ban VARCHAR(50) NOT NULL,
  nom_cuent_Ban VARCHAR(50) NOT NULL,
  mun_cuent_Ban VARCHAR(50) NOT NULL
);

CREATE TABLE categoria (
  id_categoria SERIAL PRIMARY KEY,
  nom_categoria VARCHAR(450) NOT NULL
);

-- 2. Tablas que dependen de pais
CREATE TABLE provincia (
  id_provin SERIAL PRIMARY KEY,
  nom_provin VARCHAR(100) NOT NULL,
  id_pais INTEGER,
  FOREIGN KEY (id_pais) REFERENCES pais(id_pais)
);

-- 3. Tablas que dependen de provincia
CREATE TABLE ciudad (
  id_ciud SERIAL PRIMARY KEY,
  nom_ciud VARCHAR(100) NOT NULL,
  id_provin INTEGER,
  FOREIGN KEY (id_provin) REFERENCES provincia(id_provin)
);

-- 4. Tablas que dependen de ciudad y estado
CREATE TABLE cliente (
  id_pers SERIAL PRIMARY KEY,
  cedr_cli VARCHAR(15) NOT NULL,
  tipo_cedr_cli VARCHAR(20) NOT NULL,
  nacion_cli VARCHAR(30) NOT NULL,
  nom_cli VARCHAR(100) NOT NULL,
  ape_cli VARCHAR(100) NOT NULL,
  fecha_naci_cli VARCHAR(12) NOT NULL,
  lugar_naci_cli VARCHAR(100) NOT NULL,
  tel_pers VARCHAR(10),
  cel_pers VARCHAR(10),
  email_pers VARCHAR(60) NOT NULL,
  edad_pers VARCHAR(20) NOT NULL,
  sexo_cli VARCHAR(20) NOT NULL,
  estado_civil_pers VARCHAR(30) NOT NULL,
  estatura_cli VARCHAR(30) NOT NULL,
  peso_cli VARCHAR(30) NOT NULL,
  parroq_cli VARCHAR(100) NOT NULL,
  calle_princ_pers VARCHAR(100) NOT NULL,
  calle_secun_pers VARCHAR(100) NOT NULL,
  id_ciud INTEGER,
  FOREIGN KEY (id_ciud) REFERENCES ciudad(id_ciud),
  id_estado INTEGER,
  FOREIGN KEY (id_estado) REFERENCES estado(id_estado)
);

-- 5. Tablas que dependen de roles y cliente
CREATE TABLE usuarios (
  id_usuario SERIAL PRIMARY KEY,
  users VARCHAR(100) NOT NULL,
  pass VARCHAR(150) NOT NULL,
  id_persona INTEGER,
  id_rol INTEGER,
  FOREIGN KEY (id_rol) REFERENCES roles(id_rol)
);

-- 7. Beneficios y dependencias
CREATE TABLE beneficios (
  id_beneficios SERIAL PRIMARY KEY,
  nom_beneficios VARCHAR(450) NOT NULL,
  id_categoria INTEGER,
  FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
);

-- 8. Tipos de seguro
CREATE TABLE tipo_seguro (
  id_tip_seg SERIAL PRIMARY KEY,
  nom_tip_seg VARCHAR(450) NOT NULL,
  descrip_tip_seg VARCHAR(450) NOT NULL,
  pago_tip_seg VARCHAR(450) NOT NULL,
  suma_tip_seg VARCHAR(450) NOT NULL,
  id_estado INTEGER,
  FOREIGN KEY (id_estado) REFERENCES estado(id_estado)
);

-- 9. Relación tipo_seguro-beneficios
CREATE TABLE seguro_bedeficio (
  id_tip_seg INTEGER,
  id_beneficios INTEGER,
  FOREIGN KEY (id_tip_seg) REFERENCES tipo_seguro(id_tip_seg),
  FOREIGN KEY (id_beneficios) REFERENCES beneficios(id_beneficios)
);

CREATE TABLE agente (
  id_agente SERIAL PRIMARY KEY,
  ced_agente VARCHAR(10) NOT NULL,
  nom_agente VARCHAR(70) NOT NULL,
  ape_agente VARCHAR(70) NOT NULL,
  email_agente VARCHAR(70) NOT NULL,
  dire_agente VARCHAR(120) NOT NULL,
  tel_agente VARCHAR(10) NOT NULL,
  id_estado INTEGER,
  FOREIGN KEY (id_estado) REFERENCES estado(id_estado)
);

-- 10. Tabla seguros con múltiples FK
CREATE TABLE seguros (
  id_seguro SERIAL PRIMARY KEY,
  ciud_seguro VARCHAR(100) NOT NULL,
  dia_seguro VARCHAR(100) NOT NULL,
  mes_seguro VARCHAR(100) NOT NULL,
  anio_seguro VARCHAR(100) NOT NULL,
  monto_seguro VARCHAR(100) NOT NULL,
  tiempo_seguro VARCHAR(100) NOT NULL,
  id_emple INTEGER,
  cargoemple VARCHAR(100) NOT NULL,
  id_pers INTEGER,
  id_tip_seg INTEGER,
  id_pers_fac INTEGER,
  id_cuent_Ban INTEGER,
  id_estado INTEGER,
  FOREIGN KEY (id_estado) REFERENCES estado(id_estado),
  FOREIGN KEY (id_pers) REFERENCES cliente(id_pers),
  FOREIGN KEY (id_tip_seg) REFERENCES tipo_seguro(id_tip_seg),
  FOREIGN KEY (id_pers_fac) REFERENCES persona_fac(id_pers_fac),
  FOREIGN KEY (id_cuent_Ban) REFERENCES cuenta_banco(id_cuent_Ban)
);

-- 11. Tabla dependientes
CREATE TABLE dependientes (
  id_depen SERIAL PRIMARY KEY,
  cedr_depen VARCHAR(15) NOT NULL,
  tipo_cedr_depen VARCHAR(20) NOT NULL,
  nom_depen VARCHAR(100) NOT NULL,
  ape_depen VARCHAR(100) NOT NULL,
  fecha_naci_depen VARCHAR(12) NOT NULL,
  parent_depen VARCHAR(70) NOT NULL,
  discap_depen VARCHAR(170) NOT NULL,
  cond_depen VARCHAR(170) NOT NULL,
  fecha_fin_cond VARCHAR(20) NOT NULL,
  fecha_ini_cond VARCHAR(20) NOT NULL,
  id_seguro INTEGER,
  FOREIGN KEY (id_seguro) REFERENCES seguros(id_seguro)
);

-- 13. Validación de email para clientes
CREATE TABLE validar_email (
  id_val_email SERIAL PRIMARY KEY,
  url_emal VARCHAR(20) NOT NULL,
  token_val_email VARCHAR(512) NOT NULL,
  id_pers INTEGER,
  FOREIGN KEY (id_pers) REFERENCES cliente(id_pers)
);

-- 14. Agente y tokens


CREATE TABLE token_agente (
  id_val SERIAL PRIMARY KEY,
  url_emal VARCHAR(20) NOT NULL,
  token_val VARCHAR(512) NOT NULL,
  id_agente INTEGER,
  FOREIGN KEY (id_agente) REFERENCES agente(id_agente)
);
